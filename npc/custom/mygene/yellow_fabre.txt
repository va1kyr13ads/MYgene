// MYgene Yellow Fabre — standalone server event.
// 30 Fabre-family kills (Fabre 1007, Pupa 1008, Creamy 1018) on ANY map
// share one server-wide counter for all players. Yellow Fabre (31050)
// appears on the map of the 30th eligible kill; its own defeat counts as
// 1 toward the NEXT 30-kill cycle. Only one Yellow Fabre exists at a time.
// Custom rewards: exact per-kill chances, independent of 4x global drop rate;
// vanilla Fabre drops come from db/import/mob_db.yml and follow server rates.
-	script	MYgene_Yellow_Fabre	-1,{
OnInit:
    .FabreKills = 0;
    .YellowActive = 0;
    end;

OnNPCKillEvent:
    // While Yellow Fabre is alive, further family kills do not advance it.
    if (.YellowActive)
        end;
    // Count eligible monster IDs on any map; do not restrict to one field.
    // Yellow Fabre (31050) is credited on OnYellowFabreDead instead.
    if (killedrid != 1007 && killedrid != 1008 && killedrid != 1018)
        end;
    .FabreKills++;
    if (.FabreKills < 30)
        end;
    .FabreKills = 0;
    .YellowActive = 1;
    // Spawn where the world-wide counter hit 30, not at a fixed field.
    .@map$ = strcharinfo(3);
    monster .@map$,0,0,"Yellow Fabre",31050,1,"MYgene_Yellow_Fabre::OnYellowFabreDead";
    mapannounce .@map$,"A Yellow Fabre has appeared after 30 Fabre-family defeats!",bc_map,0xFFFF00;
    end;

OnYellowFabreDead:
    .YellowActive = 0;
    // Yellow Fabre is itself family: its defeat counts as kill 1/30
    // toward the NEXT cycle, without spawning another Yellow Fabre here.
    .FabreKills = 1;
    // An OnMobDeath event carries the killer's RID.
    // Place rolls on the ground at the killer's position near the fallen mob.
    if (getmapxy(.@map$,.@x,.@y,BL_PC) != 0)
        end;

    // Fabre Slayer Knife: exactly 0.05% (500/1,000,000; 1 in 2,000).
    if (rand(1000000) < 500)
        makeitem 31001,1,.@map$,.@x,.@y;

    // One random Red, Blue, or Yellow Gemstone: 70.00%.
    if (rand(1000000) < 700000) {
        switch(rand(3)) {
        case 0: makeitem 715,1,.@map$,.@x,.@y; break;
        case 1: makeitem 716,1,.@map$,.@x,.@y; break;
        default: makeitem 717,1,.@map$,.@x,.@y; break;
        }
    }

    // Yellow Fabre Card: precisely 0.01% (one in ten thousand).
    if (rand(1000000) < 100)
        makeitem 31002,1,.@map$,.@x,.@y;

    // Gold: 55.00% (existing item 969, not Zeny).
    if (rand(1000000) < 550000)
        makeitem 969,1,.@map$,.@x,.@y;

    // Attraction Potion: 0.50% (one in two hundred).
    if (rand(1000000) < 5000)
        makeitem 31003,1,.@map$,.@x,.@y;
    end;
}
