// MYgene Yellow Fabre — standalone server event.
// Each map has its own 30-kill counter for Fabre-family monsters:
// Fabre (1007), Pupa (1008), Creamy (1018). Players on the SAME map
// contribute together. Yellow Fabre (31050) spawns on that map when
// its counter reaches 30. Its defeat counts as 1 toward the next cycle.
// Multiple maps can have their own Yellow Fabre alive simultaneously.
// Yellow Fabre spawns where the 30th eligible monster dies (or by its killer).
// Custom rewards: exact per-kill chances, independent of 4x global drop rate;
// vanilla Fabre drops come from db/import/mob_db.yml and follow server rates.
-	script	MYgene_Yellow_Fabre	-1,{
OnInit:
    // Map-specific counters are temporary NPC variables (reset on reload).
    .MapCount = 0;
    end;

OnNPCKillEvent:
    // Yellow Fabre (31050) gets its next-cycle credit in its death event.
    if (killedrid != 1007 && killedrid != 1008 && killedrid != 1018)
        end;
    .@map$ = strcharinfo(3);
    if (.@map$ == "")
        end;

    // Each distinct map gets its own counter/active flag. No cross-map sum.
    .@slot = -1;
    for (.@i = 0; .@i < .MapCount; .@i++) {
        if (.MapName$[.@i] == .@map$) {
            .@slot = .@i;
            break;
        }
    }
    if (.@slot < 0) {
        .@slot = .MapCount;
        .MapCount++;
        .MapName$[.@slot] = .@map$;
        .KillCount[.@slot] = 0;
        .YellowActive[.@slot] = 0;
    }
    // A map with an active Yellow Fabre waits for that one to be killed.
    if (.YellowActive[.@slot])
        end;
    .KillCount[.@slot]++;
    if (.KillCount[.@slot] < 30)
        end;
    // Place the encounter where the 30th family member fell, not randomly
    // across the map. OnNPCKillEvent provides that slain mob's unique GID.
    // Prefer its actual tile while rAthena still has the corpse unit.
    // If the GID is already gone, use the killing player's current tile.
    .@spawn_map$ = .@map$;
    .@located = 0;
    if (killedgid > 0 && unitexists(killedgid)) {
        getunitdata killedgid,.@mob_data;
        .@death_map$ = mapid2name(.@mob_data[UMOB_MAPID]);
        if (.@death_map$ == .@map$ && .@mob_data[UMOB_X] > 0 && .@mob_data[UMOB_Y] > 0) {
            .@x = .@mob_data[UMOB_X];
            .@y = .@mob_data[UMOB_Y];
            .@located = 1;
        }
    }
    if (!.@located && getmapxy(.@spawn_map$,.@x,.@y,BL_PC) != 0)
        end;
    .KillCount[.@slot] = 0;
    .YellowActive[.@slot] = 1;
    monster .@map$,.@x,.@y,"Yellow Fabre",31050,1,"MYgene_Yellow_Fabre::OnYellowFabreDead";
    mapannounce .@map$,"A Yellow Fabre has appeared",bc_map,0xFFFF00;
    end;

OnYellowFabreDead:
    // Mob-death callbacks attach the killer; credit the matching map only.
    .@map$ = strcharinfo(3);
    for (.@i = 0; .@i < .MapCount; .@i++) {
        if (.MapName$[.@i] == .@map$) {
            .YellowActive[.@i] = 0;
            .KillCount[.@i] = 1;
            break;
        }
    }
    // Place rewards on the ground at the killer's location.
    if (getmapxy(.@map$,.@x,.@y,BL_PC) != 0)
        end;

    // Legendary Bubble Gum (31009): for 15 seconds, every defined
    // special drop succeeds, including the exclusive weapon and card.
    // Outside that window each independent fixed chance is unchanged.
    .@guaranteed = (MYGENE_LEGENDARY_GUM_UNTIL > gettimetick(2));

    // Fabre Slayer Knife: exactly 0.05% (500/1,000,000; 1 in 2,000).
    // Deliver successful rolls to the killer so rare weapons are not missed
    // on the ground. Keep a floor-drop fallback for full inventories.
    if (.@guaranteed || rand(1000000) < 500) {
        if (checkweight(31001,1))
            getitem 31001,1;
        else
            makeitem 31001,1,.@map$,.@x,.@y;
    }

    // One random Red, Blue, or Yellow Gemstone: 70.00%.
    if (.@guaranteed || rand(1000000) < 700000) {
        switch(rand(3)) {
        case 0: makeitem 715,1,.@map$,.@x,.@y; break;
        case 1: makeitem 716,1,.@map$,.@x,.@y; break;
        default: makeitem 717,1,.@map$,.@x,.@y; break;
        }
    }

    // Yellow Fabre Card: precisely 0.01% (one in ten thousand).
    if (.@guaranteed || rand(1000000) < 100)
        makeitem 31002,1,.@map$,.@x,.@y;

    // Gold: 11.00% (existing item 969, not Zeny).
    if (.@guaranteed || rand(1000000) < 110000)
        makeitem 969,1,.@map$,.@x,.@y;

    // Attraction Potion: 0.50% (one in two hundred).
    if (.@guaranteed || rand(1000000) < 5000)
        makeitem 31003,1,.@map$,.@x,.@y;
    end;
}
