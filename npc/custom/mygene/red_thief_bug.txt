// MYgene Red Thief Bug: Thief Bug-family counterpart to Yellow Fabre.
// Eligible family: Thief Bug Egg (1048), Thief Bug (1051),
// Thief Bug Female (1053), Thief Bug Male (1054).
// 30 qualifying kills on ONE map spawn Red Thief Bug at the final kill.
// Per-map state is independent. An active Red Thief Bug blocks another
// spawn on that map until killed; its death counts as one next-cycle kill.
// Fixed exclusive rewards mirror Yellow Fabre and do not use global drop rates.
// Normal Thief Bug material drops are in db/import/mob_db.yml.
-	script	MYgene_Red_Thief_Bug	-1,{
OnInit:
    .MapCount = 0;
    end;

OnNPCKillEvent:
    if (killedrid != 1048 && killedrid != 1051 && killedrid != 1053 && killedrid != 1054)
        end;
    .@map$ = strcharinfo(3);
    if (.@map$ == "")
        end;

    .@slot = -1;
    for (.@i = 0; .@i < .MapCount; .@i++) {
        if (.MapName$[.@i] == .@map$) {
            .@slot = .@i;
            break;
        }
    }
    if (.@slot < 0) {
        .@slot = .MapCount;
        .MapCount++;
        .MapName$[.@slot] = .@map$;
        .KillCount[.@slot] = 0;
        .RedActive[.@slot] = 0;
    }
    if (.RedActive[.@slot])
        end;
    .KillCount[.@slot]++;
    if (.KillCount[.@slot] < 30)
        end;

    .@located = 0;
    if (killedgid > 0 && unitexists(killedgid)) {
        getunitdata killedgid,.@mob_data;
        .@death_map$ = mapid2name(.@mob_data[UMOB_MAPID]);
        if (.@death_map$ == .@map$ && .@mob_data[UMOB_X] > 0 && .@mob_data[UMOB_Y] > 0) {
            .@x = .@mob_data[UMOB_X];
            .@y = .@mob_data[UMOB_Y];
            .@located = 1;
        }
    }
    if (!.@located && getmapxy(.@map$,.@x,.@y,BL_PC) != 0)
        end;
    .KillCount[.@slot] = 0;
    .RedActive[.@slot] = 1;
    monster .@map$,.@x,.@y,"Red Thief Bug",31051,1,"MYgene_Red_Thief_Bug::OnRedThiefBugDead";
    mapannounce .@map$,"A Red Thief Bug has appeared",bc_map,0xFF5555;
    end;

OnRedThiefBugDead:
    .@map$ = strcharinfo(3);
    for (.@i = 0; .@i < .MapCount; .@i++) {
        if (.MapName$[.@i] == .@map$) {
            .RedActive[.@i] = 0;
            .KillCount[.@i] = 1;
            break;
        }
    }
    // Fixed exclusive drops; separate independent rolls with denominator
    // 1,000,000 just like the verified Yellow Fabre encounter.
    // Item drops land by the killing player's location.
    if (getmapxy(.@map$,.@x,.@y,BL_PC) != 0)
        end;

    // Legendary Bubble Gum (31009): for 15 seconds, every defined
    // special drop succeeds, including the exclusive weapon and card.
    // Outside that window each independent fixed chance is unchanged.
    .@guaranteed = (MYGENE_LEGENDARY_GUM_UNTIL > gettimetick(2));

    // Thief Bug Slayer Sword [1]: 0.05% normally (1 in 2,000).
    // When Legendary Gum is active, grant it to the attached killer
    // directly, with a floor-drop fallback when inventory cannot hold it.
    // Preserve the normal low-chance floor-drop behavior without Gum.
    if (.@guaranteed) {
        if (checkweight(31006,1)) {
            getitem 31006,1;
        } else {
            makeitem 31006,1,.@map$,.@x,.@y;
        }
    } else if (rand(1000000) < 500) {
        makeitem 31006,1,.@map$,.@x,.@y;
    }

    // One random Red / Blue / Yellow Gemstone: 70.00%.
    if (.@guaranteed || rand(1000000) < 700000) {
        switch(rand(3)) {
        case 0: makeitem 715,1,.@map$,.@x,.@y; break;
        case 1: makeitem 716,1,.@map$,.@x,.@y; break;
        default: makeitem 717,1,.@map$,.@x,.@y; break;
        }
    }

    // MYgene Red Thief Bug Card: 0.01% (1 in 10,000).
    if (.@guaranteed || rand(1000000) < 100) {
        makeitem 31007,1,.@map$,.@x,.@y;
    }

    // Gold item 969, not Zeny: 11.00%.
    if (.@guaranteed || rand(1000000) < 110000) {
        makeitem 969,1,.@map$,.@x,.@y;
    }

    // Same Attraction Potion as Yellow Fabre: 0.50%.
    if (.@guaranteed || rand(1000000) < 5000) {
        makeitem 31003,1,.@map$,.@x,.@y;
    }
    end;
}
