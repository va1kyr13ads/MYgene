//===== MYgene World Relic Keeper ============================================
//= Prontera NPC for MYgene-exclusive world relics.
//= VISUAL: adjust the map, X/Y, facing and NPC sprite on the next line.
//= Current appearance uses the existing RO "4_M_MINISTER" sprite. It does not
//= overwrite existing client sprites or require new GRF resources.
//=
//= SAFETY: Stage 1 supports a GM-ONLY Perfect knife conversion for testing.
//= The world cap, soul pipeline, lease ledger and renewal are still disabled.
//= Never open conversion to players before implementing the atomic world cap.
//= Conversion rejects bound, refined, rental, graded and optioned items; it
//= MUST be tested on a backed-up development database first.
//===========================================================================

prontera,167,187,4	script	World Relic Keeper#MYgene	4_M_MINISTER,{
    mes "^C9A13C[World Relic Keeper]^000000";
    mes "A relic's power belongs to";
    mes "the living world, not to";
    mes "forgotten vaults.";
    next;

    switch(select("About World Relics:Check My Materials:Renew a World Relic:World Relic Registry:Soul Hunting Rules:Perfect Knife Upgrade [GM]:Leave")) {
    case 1:
        mes "^C9A13C[World Relic Keeper]^000000";
        mes "Perfect Fabre Slayer Knife";
        mes "is a legendary world relic.";
        mes "It is forged from a Fabre";
        mes "Slayer Knife and the rare";
        mes "Yellow Fabre Card.";
        next;
        mes "^C9A13C[World Relic Keeper]^000000";
        mes "Its intended traits are:";
        mes "^3399FF25 ATK, Weapon Lv. 4^000000";
        mes "Required Level: 40";
        mes "Range: 2, Weight: 100";
        mes "90% Fabre-family execution";
        mes "Indestructible / No Refine";
        next;
        mes "^C9A13C[World Relic Keeper]^000000";
        mes "The relic will have a";
        mes "60-day ownership lease.";
        mes "Renewals have no limit.";
        mes "Renew in the last 14 days";
        mes "or during 7 grace days.";
        close;

    case 2:
        mes "^C9A13C[World Relic Keeper]^000000";
        mes "Let me see what you carry.";
        mes "Fabre Slayer Knives: ^3399FF"+countitem(31001)+"^000000";
        mes "Yellow Fabre Cards: ^3399FF"+countitem(31002)+"^000000";
        mes "Perfect Knives: ^3399FF"+countitem(31004)+"^000000";
        next;
        mes "^C9A13C[World Relic Keeper]^000000";
        mes "Renewal will require";
        mes "^3399FF"+.SoulCost+" Fabre Family Souls^000000.";
        mes "The soul material and";
        mes "live ownership registry";
        mes "are not active yet.";
        close;

    case 3:
        mes "^C9A13C[World Relic Keeper]^000000";
        mes "I can renew a registered";
        mes "world relic for 60 days";
        mes "using 300 Family Souls.";
        next;
        mes "^C9A13C[World Relic Keeper]^000000";
        mes "^FF6600Renewal is not active yet.^000000";
        mes "The relic ledger must be";
        mes "connected first. I will";
        mes "not take your materials.";
        close;

    case 4:
        mes "^C9A13C[World Relic Keeper]^000000";
        mes "The world relic registry";
        mes "will show each unique";
        mes "relic's active status.";
        mes "Expired leases return";
        mes "their world slots.";
        next;
        mes "^C9A13C[World Relic Keeper]^000000";
        mes "^FF6600Registry not online yet.^000000";
        mes "No maximum supply has";
        mes "been configured.";
        close;

    case 5:
        mes "^C9A13C[World Relic Keeper]^000000";
        mes "Only registered owners";
        mes "will be able to collect";
        mes "Fabre Family Souls.";
        mes "The weapon need not be";
        mes "equipped for collection.";
        next;
        mes "^C9A13C[World Relic Keeper]^000000";
        mes "Planned souls per kill:";
        mes "Fabre: ^3399FF1^000000";
        mes "Pupa: ^3399FF1^000000";
        mes "Creamy: ^3399FF1^000000";
        mes "Yellow Fabre: ^3399FF5^000000";
        next;
        mes "^C9A13C[World Relic Keeper]^000000";
        mes "Only valid field kills";
        mes "can earn souls. GM and";
        mes "test spawns do not count.";
        mes "Gather ^3399FF300 Souls^000000 for";
        mes "each 60-day renewal.";
        mes "^FF6600Collection is pending.^000000";
        close;

    case 6:
        // This is deliberately NOT a public-world-limited creation flow.
        // The live registry will replace this controlled test conversion.
        mes "^C9A13C[World Relic Keeper]^000000";
        if (getgroupid() != 99) {
            mes "^FF6600Upgrades are not open yet.^000000";
            mes "The world relic registry";
            mes "must be completed first.";
            close;
        }
        if (BaseLevel < 40) {
            mes "A Perfect weapon requires";
            mes "^3399FFBase Level 40^000000.";
            mes "You may keep using your";
            mes "current knife until then.";
            close;
        }
        if (getequipid(EQI_HAND_R) != 31001 || getequipcardid(EQI_HAND_R,0) != 31002) {
            mes "Equip your original";
            mes "Fabre Slayer Knife with";
            mes "Yellow Fabre Card inside";
            mes "its first slot.";
            mes "No extra card is needed.";
            close;
        }
        // Locate the exact equipped inventory instance. Refuse special
        // instances that getitem2 cannot faithfully copy during conversion.
        getinventorylist;
        .@idx = -1;
        .@special = 0;
        for (.@i = 0; .@i < @inventorylist_count; .@i++) {
            if (@inventorylist_id[.@i] == 31001 &&
                (@inventorylist_equip[.@i] & EQP_HAND_R) != 0 &&
                @inventorylist_card1[.@i] == 31002) {
                .@idx = @inventorylist_idx[.@i];
                if (@inventorylist_refine[.@i] != 0 ||
                    @inventorylist_identify[.@i] != 1 ||
                    @inventorylist_attribute[.@i] != 0 ||
                    @inventorylist_bound[.@i] != 0 ||
                    @inventorylist_expire[.@i] != 0 ||
                    @inventorylist_enchantgrade[.@i] != 0 ||
                    @inventorylist_card2[.@i] != 0 ||
                    @inventorylist_card3[.@i] != 0 ||
                    @inventorylist_card4[.@i] != 0 ||
                    @inventorylist_option_id1[.@i] != 0 ||
                    @inventorylist_option_id2[.@i] != 0 ||
                    @inventorylist_option_id3[.@i] != 0 ||
                    @inventorylist_option_id4[.@i] != 0 ||
                    @inventorylist_option_id5[.@i] != 0)
                    .@special = 1;
                break;
            }
        }
        if (.@idx < 0 || .@special) {
            mes "This knife cannot be";
            mes "converted by the test";
            mes "service. No item changed.";
            close;
        }
        if (!checkweight(31004,1)) {
            mes "Not enough carry weight.";
            mes "Free some weight first.";
            close;
        }
        mes "Convert this carded knife";
        mes "to the ^3399FFPerfect^000000 version?";
        mes "ATK 25 / Weapon Lv. 4";
        mes "Range 2 / Required Lv. 40";
        mes "Weight 100 / No Refine";
        mes "Indestructible / 90%";
        next;
        if (select("Upgrade equipped knife:Cancel") != 1) close;
        // Revalidate after player confirmation. The character-level gate
        // above and the GM check both still apply.
        if (getgroupid() != 99 || BaseLevel < 40 ||
            getequipid(EQI_HAND_R) != 31001 ||
            getequipcardid(EQI_HAND_R,0) != 31002 ||
            !checkweight(31004,1)) {
            mes "Conditions changed.";
            mes "No item was modified.";
            close;
        }
        getinventorylist;
        .@idx = -1;
        .@special = 0;
        for (.@i = 0; .@i < @inventorylist_count; .@i++) {
            if (@inventorylist_id[.@i] == 31001 &&
                (@inventorylist_equip[.@i] & EQP_HAND_R) != 0 &&
                @inventorylist_card1[.@i] == 31002) {
                .@idx = @inventorylist_idx[.@i];
                if (@inventorylist_refine[.@i] != 0 ||
                    @inventorylist_identify[.@i] != 1 ||
                    @inventorylist_attribute[.@i] != 0 ||
                    @inventorylist_bound[.@i] != 0 ||
                    @inventorylist_expire[.@i] != 0 ||
                    @inventorylist_enchantgrade[.@i] != 0 ||
                    @inventorylist_card2[.@i] != 0 ||
                    @inventorylist_card3[.@i] != 0 ||
                    @inventorylist_card4[.@i] != 0 ||
                    @inventorylist_option_id1[.@i] != 0 ||
                    @inventorylist_option_id2[.@i] != 0 ||
                    @inventorylist_option_id3[.@i] != 0 ||
                    @inventorylist_option_id4[.@i] != 0 ||
                    @inventorylist_option_id5[.@i] != 0)
                    .@special = 1;
                break;
            }
        }
        if (.@idx < 0 || .@special) {
            mes "Knife validation failed.";
            mes "No item was modified.";
            close;
        }
        // This standalone test conversion is not a SQL transaction. Admins
        // must back up their test database before using it.
        if (!unequip(EQI_HAND_R)) {
            mes "Unable to unequip knife.";
            mes "No item was consumed.";
            close;
        }
        if (!delitemidx(.@idx,1)) {
            mes "Could not remove knife.";
            mes "No upgrade was awarded.";
            close;
        }
        getitem2 31004,1,1,0,0,31002,0,0,0;
        logmes "MYgene test: converted equipped carded 31001 to 31004 [31002], non-lease.";
        mes "^C9A13C[World Relic Keeper]^000000";
        mes "^3399FFPerfect Fabre Slayer Knife!^000000";
        mes "Your original card remains";
        mes "compounded in its slot.";
        mes "This is a TEST relic.";
        mes "It has no expiry or cap.";
        close;

    default:
        mes "^C9A13C[World Relic Keeper]^000000";
        mes "May your relic never;
        mes "fall into silence.";
        close;
    }

OnInit:
    // Preview values. The future backend must enforce these server-side.
    .SoulCost = 300;
    .LeaseDays = 60;
    .RenewalWindowDays = 14;
    .GraceDays = 7;
    .PerfectKnifeID = 31004; // Standalone GM test item, no lease/cap.
    .SoulItemID = 31005;    // Reserved; not currently in item database.
    end;
}
